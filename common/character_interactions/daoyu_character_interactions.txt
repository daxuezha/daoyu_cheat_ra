# 【筹划意外（流亡者版）】专门给流亡池角色的独立互动。
#   为什么单独做一个：原版 daoyu_arrange_accident 的 potential 里有
#   is_in_exile_pool = no，流亡者看不到它。为了**完全不动原版互动**，
#   这里另起一个只对 is_in_exile_pool = yes 成立的同名功能。
#
#   对流亡者而言「流放」没有意义（他已经在流亡池里），所以本补丁不为流亡者
#   提供原版那个「流放异见分子」按钮，只给出「筹划意外」作为唯一的处置手段。
#
#   效果本体与原版完全一致：效果代码里只用到 this（角色），不依赖 owner，
#   因此对流亡者天然可用（原版 exiles_events.txt 也直接对流亡角色 kill_character）。
#   已知边界：is_immortal = yes 的角色会被引擎忽略（点了没反应），与原版一致。

# 【授予指挥权】政治家 / 鼓动者 -> 陆军将领
#   原版只有 grant_command_to_ruler（统治者）与 grant_command_to_agitator（鼓动者），
#   缺「政治家」这一路，且原版两个都不检查刀鱼是否开启。这里补齐四组「授予 / 剥夺」，
#   让政治家与鼓动者都能针对性地加/去陆军将领或海军提督。
#
# 【关键】剥夺指挥权**不能**用原版的 retire_commander：
#   它的 effect 里是 replace_character_roles = { remove = { general admiral } }
#   再 retire_character_if_should_be_culled = yes，会把角色**直接淘汰出人物池**
#   （表现就是「这个人之后在角色界面再也找不到、也不能操作了」）。
#   本补丁改用 add_character_role / remove_character_role **只动这一个职位**，
#   角色始终留在人物池里，可以在管理角色界面继续操作。
#
# 角色名一律裸写（不能带引号，否则静默失败）。
daoyu_ra_grant_army_command = {
	clicksound = "event:/SFX/UI/Military/command_grant"
	potential = {
		is_adult = yes
		is_in_void = no
		is_in_exile_pool = no
		is_ruler_of_own_country = no
		not = { has_role_of_type = general }
		or = { has_role_of_type = politician has_role_of_type = agitator }
		scope:actor ?= { is_ai = no has_global_variable = daoyu_on }
	}
	possible = {
		is_female = no
		owner ?= { exists = owner.capital.land_hq }
	}
	show_requirements = yes
	effect = {
		add_character_role = general
	}
}

daoyu_exile_character = {
	clicksound = "event:/SFX/DLC/1.3_ip1/UI/exile_character"
	potential = {
		owner ?= { is_ai = no }
		has_global_variable = daoyu_on
		is_in_exile_pool = no
		is_in_void = no
		is_ruler = no
		is_heir = no
		or = {
			has_role_of_type = politician
			has_role_of_type = agitator
		}
	}
	show_requirements = yes
	effect = {
		this.interest_group = {
			save_scope_as = relevant_ig
		}
		this = {
			save_scope_as = exile_scope
		}
		owner = {
			save_scope_as = exiling_country_scope
		}
		if = {
			limit = {
				has_role_of_type = politician
			}
			owner = {
				scope:relevant_ig = {
					add_modifier = {
						name = modifier_exiled_leader
						days = normal_modifier_time
						is_decaying = yes
					}
				}
			}
		}
		if = {
			limit = {
				owner = { has_journal_entry = je_caudillo }
				interest_group = {
					is_interest_group_type = ig_armed_forces
				}
			}
			if = {
				limit = {
					interest_group = {
						leader = {
							this = ROOT
						}
					}
				}
				owner = {
					custom_tooltip = {
						text = lose_large_caudillo_points
						change_variable = {
							name = caudillo_var
							add = -200
						}
					}
				}
			}
			else = {
				owner = {
					custom_tooltip = {
						text = lose_caudillo_points
						change_variable = {
							name = caudillo_var
							add = -100
						}
					}
				}
			}
		}
		if = {
			limit = {
				owner = {
					is_player = yes
					NOT = { has_variable = achievement_agitate_elsewhere_var }
				}
				popularity >= 100
			}
			owner = { set_variable = achievement_agitate_elsewhere_var }
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_role_of_type = agitator }
				}
				add_character_role = agitator
			}
			if = {
				limit = {
					has_role_of_type = general
				}
				remove_character_role = general
			}
			if = {
				limit = {
					has_role_of_type = admiral
				}
				remove_character_role = admiral
			}
			if = {
				limit = {
					has_role_of_type = politician
				}
				set_as_interest_group_leader = no
				remove_character_role = politician
			}
		}
		owner = {
			custom_tooltip = exiling_dissident_radicals_factors
		}
		# Extra radical from exiling a popular character
		if = {
			limit = {
				popularity >= 50
			}
			owner = {
				add_radicals = {
					value = small_radicals
					interest_group = scope:relevant_ig
				}
			}
		}
		# Standard radicals from having Right of Assembly
		if = {
			limit = {
				owner = { has_law = law_type:law_right_of_assembly }
			}
			owner = {
				add_radicals = {
					value = medium_radicals
					interest_group = scope:relevant_ig
				}
			}
		}
		# Fewer radicals from Censorship
		else_if = {
			limit = {
				owner = { has_law = law_type:law_censorship }
			}
			owner = {
				add_radicals = {
					value = small_radicals
					interest_group = scope:relevant_ig
				}
			}
		}
		# Boring characters should be made interesting when being exiled
		if = {
			limit = {
				has_ideology = ideology:ideology_moderate
			}
			adopt_random_antigovernment_ideology = yes
		}
		hidden_effect = {
			if = {
				limit = {
					popularity > 15
				}
				add_modifier = {
					name = in_exile_modifier
				}
				every_country = {
					limit = {
						has_diplomatic_relevance = scope:exiling_country_scope
					}
					post_notification = exile_notification
				}
			}
			else = {
				add_modifier = {
					name = in_exile_modifier
					multiplier = 3
				}
			}
		}
		custom_tooltip = {
			text = exile_dissident_tt
			exile_character_with_martyrdom = yes
		}
	}
	show_confirmation_box = yes
}

daoyu_interactions_manage_character = {
	clicksound = "event:/SFX/DLC/1.3_ip1/UI/exile_character"
	potential = {
		# 【兼容补丁新增】?= 的语义是「作用域存在 且 满足条件」。
		# 流亡池角色没有 owner，原来的 owner ?= { is_ai = no } 会返回假，
		# 导致「管理角色」对流亡者根本不显示。这里补上 scope:actor 这条路径
		# （流亡池场景下 scope:actor 就是发起国）。
		or = {
			owner ?= { is_ai = no }
			scope:actor ?= { is_ai = no }
		}
		has_global_variable = daoyu_on
		not = { has_variable = daoyu_dead }
	}
	possible = {
		custom_tooltip = {
			text = daoyu_manage_character_cannot_be_opened_repeatedly
			not = { has_global_variable = daoyu_character_interactions }
			not = { has_global_variable = daoyu_character_events }
		}
		custom_tooltip = {
			text = daoyu_has_been_activated
			not = { has_global_variable = daoyu_cheat }
		}
	}
	show_requirements = yes
	effect = {
		this = { add_trait = daoyu_target_character }
		set_global_variable = daoyu_character_interactions
		# 【兼容补丁新增】把「本次点选的角色」存成发起国上的变量（daoyu_ra_targeted_character），
		# 让 daoyu.220 精确作用在这一个人身上。这只是把基础 Mod 原本的
		#   random_character = { limit = { has_trait = daoyu_target_character } ... }
		# 从「随机取一个带标签的角色」变成「就是我刚才点的这个」；
		# 基础 Mod 的随机回退逻辑（daoyu.220 的 immediate）完整保留，故不改变原有行为。
		save_scope_as = daoyu_ra_clicked_character
		# 流亡池中的角色没有 owner，此时必须走 scope:actor（交互的发起国）
		if = {
			limit = { is_in_exile_pool = yes }
			scope:actor = {
				set_variable = daoyu_target_country
				set_variable = { name = daoyu_ra_targeted_character value = scope:daoyu_ra_clicked_character }
				trigger_event = { id = daoyu.220 popup = yes }
			}
		}
		else = {
			owner = {
				set_variable = daoyu_target_country
				set_variable = { name = daoyu_ra_targeted_character value = scope:daoyu_ra_clicked_character }
				trigger_event = { id = daoyu.220 popup = yes }
			}
		}
	}
}

# Daoyu Cheat - Real History Compatibility patch additions
# 本补丁新增的角色交互（追加于文件末尾，不改动基础 Mod 既有内容）
#
# 术语区分（重要）：
#   邀请 Invite   —— 用于【鼓动者】：把流亡池中的鼓动者请进本国鼓动者槽位，
#                     并让其参加 / 发起政治运动。
#   招募 Recruit  —— 用于【政府职位】：让角色担任贵族、政治家、指挥官等职务。
#   转职 Convert  —— 把已有职位（政治家 / 指挥官）的角色改成鼓动者。
#
# 原版限制说明：
#   invite_exile 带有 cooldown = { days = normal_modifier_time }（1825 天 = 5 年），
#   且 possible 中有 can_agitate = scope:actor。这就是「最近 5 年招募过政治鼓动者
#   后不得再招募」的来源。本补丁的邀请交互不声明 cooldown、也不检查 can_agitate。

#   先补齐鼓动者槽位，再绕过 5 年冷却把人请进来，并自动确保其有可参加的政治运动。
